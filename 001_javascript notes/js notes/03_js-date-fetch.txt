 <script>
       
     function getTimeandHours(){
         let mydate = new Date();
       let h = mydate.getHours();
       let m = mydate.getMinutes();
       let s = mydate.getSeconds();
       document.getElementById("hour").innerHTML = h +"H";
       document.getElementById("mins").innerHTML = m +"M";
       document.getElementById("secs").innerHTML = s +"S";
     }

     setInterval(()=>{
        getTimeandHours();
     },1000)
      


 </script>
